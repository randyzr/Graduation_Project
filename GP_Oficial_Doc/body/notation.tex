\chapter*{Notación}
\addcontentsline{toc}{chapter}{Notación}

\begin{description}
    \item[$R$] Resistencia eléctrica.
    
    \item[$I$] Corriente eléctrica.
    
    \item[$V$] Tensión eléctrica.
    
    \item[$Q$] Carga eléctrica.
    
    \item[$C$] Capacitancia.
    
    \item[$Z$] Impedancia eléctrica.
    
    \item[$X$] Reactancia eléctrica.
    
    \item[$f$] Frecuencia de la señal eléctrica.
    
    \item[$T$] Temperatura.
    
    \item[$R_{Body}$] Resistencia eléctrica del cuerpo del usuario a tierra.
    
    \item[$V_{Body}$] Tensión medida en el cuerpo del usuario durante la prueba.
    
    \item[$I_{Body}$] Corriente que circula a través del cuerpo del usuario durante la prueba.
    
    \item[$R_{ref}$] Resistencia de referencia utilizada en el circuito de medición.
    
    \item[$V_s$] Tensión de estímulo aplicada al circuito de medición.
    
    \item[$I_{fuga}$] Corriente de fuga presente en el circuito o en la tarjeta electrónica.

\end{description}
